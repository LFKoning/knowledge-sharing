# ------------------------------- #
# fetch & pull                    #
# ------------------------------- #

# Clone the pig latin repo.
git clone https://github.com/LFKoning/pig-latin

# Personalize pyproject.toml.
# Add author name and e-mail.
code pyproject.toml

# Commit the change.
git add -A
git commit -m "Personalize project"

# Personalize on remote Github repo.

# Check local main is personalized.
git log

# Check origin/main; still shows only initial commit.
git log origin/main

# Fetch changes from the remote.
# No impact on working directory!
git fetch

# Note the remote personalization appeared.
git log origin/main

# Pull the changes into the working directory.
# Creates a merge conflict!
git pull

# Note that brances have diverged!
git status

# Fix conflict as per usual.
code pyproject.toml

git add -A
git commit -m "Fix personalization"

# ------------------------------- #
# remote add                      #
# ------------------------------- #

# Use the repository created earlier.
# Using origin as name is standard.
git remote add personal https://github.com/LFKoning/personal

# Show remote was added succesfully.
git remote -v

# Push and set up tracking on origin/main.
git push personal

# Can add another remote.
git remote set-url origin https://github.com/LFKoning/personal

# Show both remotes.
git remote -v

# Update README.md repo URLs.
code README.md

# Now we can just use push.
git push

# To delete a remote.
git remote remove personal

git remote -v


# ------------------------------- #
# branch tracking                 #
# ------------------------------- #

# Only main branch
git branch

# Create a new local branch
git checkout -b test-tracking

# Show new branch.
git branch

# No upstream branch to track...
git push

# Show tracking info using -vv.
git branch -vv

# Push and set tracking to origin/test-tracking.
git push --set-upstream origin test-tracking

# Show updated tracking info.
git branch -vv

# Show local and remote with tracking info.
git branch -avv

# Remote info is under .git/refs/remotes
dir .\.git\refs\remotes\
dir .\.git\refs\remotes\origin


# ------------------------------- #
# delete branch                   #
# ------------------------------- #

# Switch to main branch.
git checkout main

git branch -a

# Delete a local branch.
git branch -d test-tracking

# Local branch gone, but not remote.
git branch -a

# Remote requires a push.
git push origin --delete test-tracking

# Verify it is gone.
git branch -a
